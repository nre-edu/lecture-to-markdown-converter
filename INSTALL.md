```bash
cp -r "$SRC/skills/lecture-to-markdown-converter" ~/.claude/skills/
```

```powershell
$src = Join-Path $env:TEMP ([guid]::NewGuid())
git clone --depth 1 --branch main https://github.com/nre-nec/lecture-to-markdown-converter.git $src
New-Item -ItemType Directory -Force "$env:USERPROFILE\.claude\skills" | Out-Null
Copy-Item -Recurse "$src\skills\lecture-to-markdown-converter" "$env:USERPROFILE\.claude\skills\"
```
